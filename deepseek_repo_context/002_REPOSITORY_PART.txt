an_size(size: int) -> str:

    units = ["B", "KB", "MB", "GB", "TB"]

    value = float(size)

    for unit in units:

        if value < 1024:
            return f"{value:.1f} {unit}"

        value /= 1024

    return f"{value:.1f} PB"


def read_text(path: Path):

    try:

        data = path.read_bytes()

        # Detect common binary files.
        if b"\x00" in data:
            return None

        return data.decode("utf-8")

    except (OSError, UnicodeDecodeError):

        return None

def collect_files():

    files = []

    for path in ROOT.rglob("*"):

        if not path.is_file():
            continue

        if should_ignore(path):
            continue

        try:
            size = path.stat().st_size
        except OSError:
            continue

        relative = str(
            path.relative_to(ROOT)
        )

        # Explicit binary/media.
        if path.suffix.lower() in BINARY_EXTENSIONS:

            files.append({
                "path": relative,
                "type": "BINARY",
                "mime": get_mime(path),
                "size": size,
                "content": None,
            })

            continue

        content = read_text(path)

        if content is None:

            files.append({
                "path": relative,
                "type": "BINARY_OR_UNREADABLE",
                "mime": get_mime(path),
                "size": size,
                "content": None,
            })

        else:

            files.append({
                "path": relative,
                "type": "TEXT",
                "mime": get_mime(path),
                "size": size,
                "content": content,
            })

    files.sort(
        key=lambda item: item["path"].lower()
    )

    return files
