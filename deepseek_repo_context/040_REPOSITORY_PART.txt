def main():

    print()
    print("Scanning repository...")
    print()

    OUTPUT_DIR.mkdir(
        exist_ok=True
    )

    # --------------------------------------------------------
    # COLLECT FILES
    # --------------------------------------------------------

    files = collect_files()

    print(
        f"Found {len(files):,} relevant files."
    )

    # --------------------------------------------------------
    # MANIFEST
    # --------------------------------------------------------

    tree = build_tree(files)

    manifest = create_manifest(
        files,
        tree
    )

    manifest_path = (
        OUTPUT_DIR /
        "000_MANIFEST.txt"
    )

    manifest_path.write_text(
        manifest,
        encoding="utf-8"
    )

    print(
        f"Created 000_MANIFEST.txt: "
        f"{len(manifest):,} characters"
    )

    # --------------------------------------------------------
    # PART STATE
    # --------------------------------------------------------

    part_number = 1

    header = part_header(
        part_number
    )

    current = header

    # --------------------------------------------------------
    # PROCESS FILES
    # --------------------------------------------------------

    for item in files:

        # ====================================================
        # BINARY FILE
        # ====================================================

        if item["type"] != "TEXT":

            block = binary_representation(
                item
            )

            # If it doesn't fit in the current part,
            # write the current part first.
            if (
                len(current) +
                len(block)
                > MAX_CHARS
            ):

                # Only write if it contains something
                # besides the header.
                if len(current) > len(header):
                    write_part(
                        part_number,
                        current[len(header):]
                    )
